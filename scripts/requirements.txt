# Python dependencies of this repo's tool scripts.
#
#   pip install -r requirements.txt
#
# Determined by parsing the imports of every script in the repo root, not from
# memory. The "used by" count says how many scripts need the package - the
# previous version named pyserial only, which left pyOCD to a separate path and
# made "pip install -r requirements.txt" an incomplete instruction.

# Serial console: cli.py, smoketest.py, test_lan8651.py, test_mirror.py (used by: 4)
pyserial==3.5

# Flashing and register access over the EDBG probe: flash_same54.py,
# fuses_same54.py, probes.py, install_prereqs.py                       (used by: 4)
# Brings intelhex along, which flash_same54.py and fuses_same54.py import for
# HEX images - pyOCD pins its own version, so it gets no line of its own here.
#
# PINNED, and not as a precaution ritual: 0.44.1 breaks the SWD connection on
# this setup. The same version is PYOCD_PIN in install_prereqs.py - change it
# here and you must change it there too. install.bat / setup.bat step 3 still
# install it that way; this line only makes the plain pip route complete.
pyocd==0.43.0

# sv-ttk theme (Windows 11 Fluent look): bridge_gui.py, gui_term.py (used by: 2)
# A hard dependency of both, not optional - they always run themed now, there
# is no plain-ttk fallback build anymore (the separate "_modern" comparison
# builds were merged into these two files, see docs/FALLSTRICKE.md 2026-08-28).
sv-ttk==2.6.1

# X.509 cert/key generation and inspection: pki.py, mqtt_broker.py (used by: 2)
# Was already a de-facto dependency (pki.py has imported it since it was
# written) but missing from this file until now - a fresh
# "pip install -r requirements.txt" was silently incomplete.
cryptography==50.0.1

# TLS-only MQTT broker (docs/mqtt-tls-agent-prompt.md): mqtt_broker.py,
# the "MQTT" tab in bridge_gui_telnet.py (used by: 2). hbmqtt, the older
# project this was forked from, is unmaintained - do not substitute it in.
amqtt==0.12.0

# Browser front end: bridge_web_telnet.py (used by: 1). Only that one script
# needs it - the Tk GUI, the CLI tools and bridge_core itself do not, so a
# checkout that never opens the web UI can skip this line.
#
# Pulls FastAPI/uvicorn and ~35 further packages in turn. It also ships its own
# xterm.js build (nicegui/elements/xterm/dist), which is why the web terminal
# needs no CDN and works on a machine with no internet access.
nicegui==3.16.0

# TEST ONLY - run_tests.bat's second half (scripts/test_web_ui.py, NiceGUI's
# simulated browser client). Nothing that ships needs these, and run_tests.bat
# skips that suite with a note rather than failing if they are absent. The first
# suite, scripts/test_bridge_core_sim.py, is stdlib-only and always runs.
pytest==9.1.1
pytest-asyncio==1.4.0
